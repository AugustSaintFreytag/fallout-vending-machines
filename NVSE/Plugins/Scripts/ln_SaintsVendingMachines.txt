; UDFs

ref OnActivateUDF = CompileScript "SaintsVendingMachines\OnActivate.gek"
ref ScanAndLockUDF = CompileScript "SaintsVendingMachines\ScanAndLock.gek"

; Legacy State

RefMapArrayDestroy "SVM_TimeLastLocked"

; Cell Enter Event

SetEventHandler "OnCellEnter" (begin function { ref rEnteredCell }
	Call ScanAndLockUDF
end)

; Post Load Game Event

SetEventHandler "PostLoadGame" (begin function { short bSuccessfullyLoaded }
	if eval !bSuccessfullyLoaded
		return
	endif

	Call ScanAndLockUDF
end)

; Activate Event

SetEventHandlerAlt "ShowOff:OnPreActivate" (begin function { ref rActivator, short bIsActivationNotPrevented }
	ref rActivated = GetSelfAlt

	if eval HasKeyword rActivated "NukaColaVendingMachine"
		if eval player.IsSneaking == 1
			return
		endif

		SetFunctionValue 0
		Call OnActivateUDF rActivated
	endif
end) 1::Player
