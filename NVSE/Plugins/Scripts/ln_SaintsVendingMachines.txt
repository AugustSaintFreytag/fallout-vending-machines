ref rEnteredCell
short iLoad

ref rActivator
ref rActivated

short bIsActivationNotPrevented

short iButton = -1

ref VendingMachineActivateUDF = CompileScript "SaintsVendingMachines\SVM_OnActivateVendingMachine.gek"
ref VendingMachineLockScanUDF = CompileScript "SaintsVendingMachines\SVM_ScanAndLock.gek"

; On Cell Enter

SetEventHandler "OnCellEnter" (begin function { ref rEnteredCell }
	Call VendingMachineLockScanUDF rEnteredCell
end)

; Post Load Game

SetEventHandler "PostLoadGame" (begin function { short iLoad }
	if iLoad
		Call VendingMachineLockScanUDF PlayerREF.GetParentCell
	endif
end)

; Activate

SetEventHandlerAlt "ShowOff:OnPreActivate" (begin function { rActivator, bIsActivationNotPrevented }
	let rActivated := GetSelfAlt

	if eval HasKeyword rActivated "NukaColaVendingMachine" || HasKeyword rActivated "SunsetSarsaparillaVendingMachine"
		if eval player.IsSneaking == 1
			return
		endif

		SetFunctionValue 0
		Call VendingMachineActivateUDF iButton
	endif
end) 1::Player
